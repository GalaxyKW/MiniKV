cmake_minimum_required(VERSION 3.16)
project(minikv_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

option(MINIKV_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(MINIKV_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()

option(MINIKV_FUZZING "Build codec fuzzing with Clang/libFuzzer and ASan/UBSan" OFF)
if(MINIKV_FUZZING)
    if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
        message(FATAL_ERROR "MINIKV_FUZZING requires Clang with libFuzzer; use make cpp-fuzz-test FUZZ_CXX=clang++")
    endif()
    add_executable(codec_fuzz ../tests/codec_fuzz.cpp)
    target_include_directories(codec_fuzz PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_options(codec_fuzz PRIVATE -O1 -g -Wall -Wextra -Wpedantic
        -fsanitize=fuzzer,address,undefined -fno-omit-frame-pointer -fno-sanitize-recover=all)
    target_link_options(codec_fuzz PRIVATE -fsanitize=fuzzer,address,undefined -fno-sanitize-recover=all)
endif()

add_library(minikv_storage engine.cpp)
target_include_directories(minikv_storage PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(minikv_storage PUBLIC Threads::Threads)
target_compile_options(minikv_storage PRIVATE -Wall -Wextra -Wpedantic)

add_executable(engine server.cpp)
target_link_libraries(engine PRIVATE minikv_storage)
target_compile_options(engine PRIVATE -Wall -Wextra -Wpedantic)

include(CTest)
if(BUILD_TESTING)
    add_executable(engine_test ../tests/engine_test.cpp)
    target_link_libraries(engine_test PRIVATE minikv_storage)
    add_test(NAME storage_recovery COMMAND engine_test)
    add_executable(snapshot_test ../tests/snapshot_test.cpp)
    target_link_libraries(snapshot_test PRIVATE minikv_storage)
    add_test(NAME snapshot_concurrency COMMAND snapshot_test)
    add_executable(stats_test ../tests/stats_test.cpp)
    target_link_libraries(stats_test PRIVATE minikv_storage)
    add_test(NAME operational_stats COMMAND stats_test)
    add_executable(async_test ../tests/async_test.cpp)
    target_link_libraries(async_test PRIVATE minikv_storage)
    add_test(NAME async_completion COMMAND async_test)
    add_executable(wal_format_test ../tests/wal_format_test.cpp)
    target_link_libraries(wal_format_test PRIVATE minikv_storage)
    add_test(NAME wal_format_upgrade COMMAND wal_format_test)
    add_executable(filesystem_test ../tests/filesystem_test.cpp)
    target_link_libraries(filesystem_test PRIVATE minikv_storage)
    # Count only reads of a selected legacy fixture to verify early rejection
    # without memory limits (which conflict with sanitizer shadow mappings).
    target_link_options(filesystem_test PRIVATE -Wl,--wrap=read)
    add_test(NAME storage_filesystem COMMAND filesystem_test)
    add_executable(data_limit_test ../tests/data_limit_test.cpp)
    target_link_libraries(data_limit_test PRIVATE minikv_storage)
    add_test(NAME data_capacity COMMAND data_limit_test)
    set_tests_properties(storage_recovery snapshot_concurrency operational_stats async_completion wal_format_upgrade storage_filesystem data_capacity PROPERTIES TIMEOUT 60)
endif()
